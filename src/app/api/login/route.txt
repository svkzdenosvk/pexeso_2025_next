// app/api/login/route.ts
import { NextResponse } from 'next/server';
import { adminAuth } from '@pexeso/lib/firebase/firebase-admin';
import { My_Type_Login } from '@pexeso/_inc/my_types';
import { getDoc, doc } from 'firebase/firestore';
import { projectUsers } from '@pexeso/lib/firebase/firestoreConfigUsers'; // Firestore z client SDK

export async function POST(req: Request) {
  try {
    const { email, password }: My_Type_Login = await req.json();

    if (!email || !password) {
      return NextResponse.json({ error: 'Missing credentials' }, { status: 400 });
    }

    // Firebase Admin SDK NEPODPORUJE login cez heslo
    // Preto musíš použiť Firebase REST API

    const res = await fetch(`https://identitytoolkit.googleapis.com/v1/accounts:signInWithPassword?key=${process.env.NEXT_PUBLIC_FIREBASE_API_KEY}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        email,
        password,
        returnSecureToken: true,
      }),
    });

    const data = await res.json();

    if (!res.ok) {
      return NextResponse.json({ error: data.error?.message || 'Login failed' }, { status: 401 });
    }

     // Získaj meno z Firestore (napr. kolekcia users, dokument s ID = uid)
    const userDoc = await getDoc(doc(projectUsers, data.localId));
    const name = userDoc.exists() ? userDoc.data().name ?? '' : '';

    // Nastav token cookie
    const response = NextResponse.json({
      user: {
        uid: data.localId,
        email: data.email,
      },
      name,
    });

    // Príklad: nastav cookie s ID tokenom
    // const response = NextResponse.json({ success: true });
    response.cookies.set('token', data.idToken, {
      httpOnly: true,
      path: '/',
      secure: process.env.NODE_ENV === 'production',
      maxAge: 60 * 60 * 24, // 1 deň
    });

    return response;
  } catch (err) {
    console.error('Login error:', err);
    return NextResponse.json({ error: 'Unexpected error' }, { status: 500 });
  }
}
